# Multi-stage build optimized for ROS2 Humble Robot Pi deployment
ARG FROM_IMAGE=ros:humble-ros-base
ARG OVERLAY_WS=/home/robot/workspace/my_steel-robot_ws

# Stage 1: Dependency caching and resolution
FROM $FROM_IMAGE AS cacher
ARG OVERLAY_WS
ARG REPO_URL=https://github.com/goldjunge91/my_steel-robot_ws.git
ARG REPO_BRANCH=humble
ARG VCS_FILE=src/ros2.repos

ENV DEBIAN_FRONTEND=noninteractive

# Optimize apt caching and disable recommends for smaller image
RUN --mount=type=cache,target=/var/cache/apt,sharing=locked \
        --mount=type=cache,target=/var/lib/apt,sharing=locked \
        rosdep update --rosdistro $ROS_DISTRO && \
        cat <<EOF > /etc/apt/apt.conf.d/docker-clean && apt-get update
APT::Install-Recommends "false";
APT::Install-Suggests "false";
EOF

# Install essential build tools and VCS tools
RUN --mount=type=cache,target=/var/cache/apt,sharing=locked \
        --mount=type=cache,target=/var/lib/apt,sharing=locked \
        apt-get install -y --no-install-recommends \
        build-essential \
        cmake \
        git \
        python3-pip \
        python3-colcon-common-extensions \
        python3-colcon-mixin \
        python3-rosdep \
        python3-vcstool \
        ca-certificates

# Create robot user for security best practices
RUN useradd -m -s /bin/bash robot && \
        usermod -aG sudo robot && \
        echo "robot ALL=(ALL) NOPASSWD:ALL" >> /etc/sudoers && \
        mkdir -p $OVERLAY_WS && \
        chown -R robot:robot /home/robot

USER robot
WORKDIR $OVERLAY_WS

# Clone repository and import VCS dependencies
RUN git clone --branch "${REPO_BRANCH}" --depth 1 "${REPO_URL}" . && \
        vcs import src < "${VCS_FILE}"

# Derive build and runtime dependencies using rosdep
RUN bash -e <<'EOF'
declare -A types=(
  [exec]="--dependency-types=exec"
  [build]="")
for type in "${!types[@]}"; do
  rosdep install -y \
    --from-paths src \
    --ignore-src \
    --reinstall \
    --simulate \
    --rosdistro $ROS_DISTRO \
    ${types[$type]} \
    --skip-keys "laser_filters gazebo_ros gazebo_ros2_control ros_gz_sim ros_gz_bridge python3-libgpiod" \
    | grep 'apt-get install' \
    | awk '{gsub(/'\''/,"",$4); print $4}' \
    | sort -u > /tmp/${type}_debs.txt
done
EOF

# Stage 2: Build environment with dependencies
FROM $FROM_IMAGE AS builder
ARG OVERLAY_WS
ARG COLCON_PARALLEL_WORKERS=2

ENV DEBIAN_FRONTEND=noninteractive
ENV COLCON_PARALLEL_WORKERS=${COLCON_PARALLEL_WORKERS}

# Install build dependencies from cacher stage
COPY --from=cacher /tmp/build_debs.txt /tmp/build_debs.txt
COPY --from=cacher /etc/apt/apt.conf.d/docker-clean /etc/apt/apt.conf.d/docker-clean
RUN --mount=type=cache,target=/var/cache/apt,sharing=locked \
        --mount=type=cache,target=/var/lib/apt,sharing=locked \
        apt-get update && \
        < /tmp/build_debs.txt xargs apt-get install -y

# Create robot user (consistent with cacher stage)
RUN useradd -m -s /bin/bash robot && \
        usermod -aG sudo robot && \
        echo "robot ALL=(ALL) NOPASSWD:ALL" >> /etc/sudoers && \
        mkdir -p $OVERLAY_WS && \
        chown -R robot:robot /home/robot

USER robot
WORKDIR $OVERLAY_WS

# Copy source code from cacher stage
COPY --from=cacher --chown=robot:robot $OVERLAY_WS/src ./src

# Build workspace with optimized colcon settings
RUN bash -c "source /opt/ros/$ROS_DISTRO/setup.bash && \
        colcon build \
        --cmake-args -DCMAKE_BUILD_TYPE=Release \
        --parallel-workers ${COLCON_PARALLEL_WORKERS} \
        --merge-install \
        --symlink-install \
        --event-handlers console_direct+ \
        --packages-up-to \
        micro_ros_agent \
        open_manipulator_x_description \
        robot_description \
        robot_hardware_interfaces \
        robot_utils \
        robot_controller \
        robot_bringup \
        robot"

# Stage 3: Runtime environment (minimal)
FROM $FROM_IMAGE AS runner
ARG OVERLAY_WS

ENV DEBIAN_FRONTEND=noninteractive
ENV ROS_DISTRO=humble
ENV ROS_DOMAIN_ID=0
ENV RMW_IMPLEMENTATION=rmw_fastrtps_cpp
ENV OVERLAY_WS=$OVERLAY_WS

# Install only runtime dependencies
COPY --from=cacher /tmp/exec_debs.txt /tmp/exec_debs.txt
COPY --from=cacher /etc/apt/apt.conf.d/docker-clean /etc/apt/apt.conf.d/docker-clean
RUN --mount=type=cache,target=/var/cache/apt,sharing=locked \
        --mount=type=cache,target=/var/lib/apt,sharing=locked \
        apt-get update && \
        < /tmp/exec_debs.txt xargs apt-get install -y && \
        rm -rf /tmp/exec_debs.txt

# Create robot user for runtime
RUN useradd -m -s /bin/bash robot && \
        usermod -aG sudo robot && \
        echo "robot ALL=(ALL) NOPASSWD:ALL" >> /etc/sudoers && \
        mkdir -p $OVERLAY_WS && \
        chown -R robot:robot /home/robot

USER robot
WORKDIR $OVERLAY_WS

# Copy only the built install artifacts
COPY --from=builder --chown=robot:robot $OVERLAY_WS/install ./install

# Setup environment to source overlay workspace
RUN echo "source /opt/ros/$ROS_DISTRO/setup.bash" >> ~/.bashrc && \
        echo "source $OVERLAY_WS/install/setup.bash" >> ~/.bashrc

# Default command sources environment and starts bash
CMD ["bash", "-c", "source /opt/ros/$ROS_DISTRO/setup.bash && source $OVERLAY_WS/install/setup.bash && bash"]# CMD ["bash","-lc","source /opt/ros/humble/setup.bash && source /home/robot/workspace/my_steel-robot_ws/install/setup.bash && bash"]
